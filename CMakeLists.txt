cmake_minimum_required(VERSION 3.16)
project(RouteLab VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(routelab src/main.cpp)
target_include_directories(routelab PRIVATE include)
add_executable(routelab-tests tests/tests.cpp)
target_include_directories(routelab-tests PRIVATE include)
if(MSVC)
  target_compile_options(routelab PRIVATE /utf-8 /W4)
  target_compile_options(routelab-tests PRIVATE /utf-8 /W4)
else()
  target_compile_options(routelab PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(routelab-tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME algorithm_tests COMMAND routelab-tests)
add_test(NAME weighted_demo COMMAND routelab solve ${CMAKE_CURRENT_SOURCE_DIR}/examples/terrain.map astar)
